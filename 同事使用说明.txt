━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 一、使用前必读
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

这是一个「网页版」看板，无需安装任何软件、不用装 Python，
用浏览器打开就能用，数据全部在你电脑本地处理，不会上传到网上。

⚠️ 前提：电脑需要能联网（首次打开会自动加载图表、表格组件）。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 二、文件说明
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

收到文件后，你会看到：

  ├── 未回销看板.html        ← 双击这个就能打开看板
  ├── 同事使用说明.txt       ← 你正在看的这个文件
  └── templates/            ← 映射表模板（可选）
      └── mapping_template.xlsx  ← 主号映射模板，复制一份填好数据就能用


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 三、如何打开看板
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

方式一：直接双击「未回销看板.html」，会用默认浏览器自动打开。

方式二：如果双击后打开的不是网页（例如变成一堆代码）：
   → 右键点击文件 → 打开方式 → 选择 Chrome / Edge / Safari 浏览器。

推荐浏览器：Chrome 或 Edge（版本较新即可）。
请勿使用老旧的 IE 浏览器。


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 四、日常使用流程
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

整体流程：打开看板 → 上传数据 → 清洗 → 筛选查看 → 导出 → 关闭

【步骤一：上传数据】

  页面左侧是「控制面板」：

  ① 点击「上传业务数据 Excel」→ 选择你的业务数据文件
     （支持 .xlsx 和 .xls 格式）

  ② 点击「上传主号映射模板 Excel」→ 选择你的映射表文件
     （可选，不传也可以运行）
     （模板在 templates/ 文件夹里，复制一份填好数据就能用）

【步骤二：执行清洗】

  点击蓝色的「🚀 执行清洗」按钮。
  等待几秒钟，右侧会出现统计结果、图表和明细表。

  ↳ 关于「基准日期列」：
    程序会自动从"出单日期 / 签单日期 / 保单日期"等列中，
    识别用来计算"到期天数"的日期列，并提示识别结果：
      ✓ 已自动识别基准日期列：XXX   → 正常，无需操作
      ⚠️ 未自动识别到基准日期列     → 页面会弹出下拉框，
        请手动选择正确的日期列，再点「🔄 重新执行清洗」

【步骤三：筛选查看】

  左侧控制面板：

  ① 先选择「🏢 所属机构」，右侧数据会自动更新
  ② 再在「🔍 业务员筛选」里勾选业务员（可一次选多个）
     → 上方搜索框可快速查找姓名，「全选 / 清空」一键操作

  右侧三个区域分别为：
  → 汇总指标：当前筛选条件下的未回销总件数
  → 未回销分布：按到期天数（0-15天、15-30天、30-60天、
     60-90天、超90天）显示各业务员件数
  → 未回销明细：详细清单，点击表头可排序
     （默认按到期天数从高到低排列）

【步骤四：导出清单】

  导出文件会直接下载到浏览器的"下载"文件夹。

  方式一：只导出选中的业务员
  → 在业务员筛选框勾选好对象
  → 选 1 人时，按钮显示「导出 姓名」，下载单个 .xlsx 文件
  → 选多人时，按钮显示「一键导出 N 人」，下载 zip 压缩包

  方式二：导出某个机构的所有业务员
  → 先选择具体机构（不要选"全部机构"）
  → 点击「导出「机构名」全部 N 人」→ 下载 zip 压缩包

  文件命名格式：时间_未回销清单_业务员姓名.xlsx

【步骤五：关闭】

  直接关闭浏览器标签页即可，无需其他操作。

 五、映射表怎么填
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

模板（templates/mapping_template.xlsx）里已有表头和示例，
照着填即可，共三列：

  A列 = 主号机构      （这个人归属的机构名）
  B列 = 主号业务员    （要替换成的"主号"姓名）
  C列 = 原始业务员名称（Excel 里的"小号 / 别名"）

示例：
  主号机构 (A) | 主号业务员 (B) | 原始业务员名称 (C)
  机构A        | 张三           | 张三小号
  机构B        | 李四           | 李小四

清洗时，程序会自动把 C 列的"小号"替换成 B 列的"主号"，
并补上 A 列的机构名。没有匹配到的业务员会保持原名，不会丢失数据。

 六、常见问题
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Q1. 打开网页后一片空白，或图表 / 表格显示不出来

   → 需要联网加载组件。可能原因：断网，或公司网络/防火墙
     拦截了组件加载。
   解决：
   ① 检查网络是否正常
   ② 切换手机热点后再刷新页面
   ③ 若公司网络拦截，可联系 IT 放行域名 cdnjs.cloudflare.com

Q2. 双击后打开的是记事本/一堆代码，而不是网页

   → 右键点击文件 → 打开方式 → 选择 Chrome 或 Edge 浏览器。
   → 或直接把文件拖进浏览器窗口。

Q3. 上传 Excel 后清洗失败或数据异常

   → 可能原因：文件损坏、被加密、或列名无法识别。
   解决：
   ① 用 Excel/WPS 重新打开文件，另存为 .xlsx 后再上传
   ② 确认文件不是只读或密码保护状态
   ③ 确认文件未被其他程序占用

Q4. 提示「未自动识别到基准日期列」

   → 说明数据里的日期列名不在常用名称里。
   解决：按页面提示，在下拉框手动选择正确的日期列，再点
   「🔄 重新执行清洗」。

Q5. 上传大文件后页面卡顿

   → 数据量过大。建议单次上传不超过 5 万行。
   → 可尝试关闭浏览器其他标签页，释放内存。

Q6. 上传的数据会不会泄露？

   → 不会。所有清洗、计算都在你本机浏览器内完成，
     数据不会上传到任何服务器。

 七、注意事项
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. 需要联网使用（首次打开会加载组件）。
  2. 推荐使用 Chrome / Edge 浏览器。
  3. 导出的文件保存在浏览器默认的"下载"文件夹中。
  4. 数据文件列名只要能对应上即可，程序会自动识别同义列名
     （例如"销售人员""业务员姓名"都会被识别为"业务员"列）。
  5. 每次打开页面都是全新状态，之前的筛选不保留，重新上传即可。
